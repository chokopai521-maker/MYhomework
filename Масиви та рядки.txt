using System;
using System.Linq;

class Program
{
    static void Main()
    {
        int[] numbers = { 2, 5, 2, 7, 8, 5, 5, 10 };

        int even = numbers.Count(x => x % 2 == 0);
        int odd = numbers.Count(x => x % 2 != 0);
        int unique = numbers.Distinct().Count();

        Console.WriteLine("Even elements: " + even);
        Console.WriteLine("Odd elements: " + odd);
        Console.WriteLine("Unique elements: " + unique);
    }
}

using System;

class Program
{
    static void Main()
    {
        int[] numbers = { 2, 5, 7, 1, 9, 4, 6 };
        Console.WriteLine("Array: " + string.Join(" ", numbers));

        Console.Write("Enter value: ");
        if (!int.TryParse(Console.ReadLine(), out int value)) return;

        int count = 0;
        foreach (int number in numbers)
        {
            if (number < value)
                count++;
        }

        Console.WriteLine("Count: " + count);
    }
}

using System;

class Program
{
    static void Main()
    {
        int[] numbers = { 7, 6, 5, 3, 4, 7, 6, 5, 8, 7, 6, 5 };
        Console.WriteLine("Array: " + string.Join(" ", numbers));

        Console.Write("Enter first number: ");
        if (!int.TryParse(Console.ReadLine(), out int a)) return;
        Console.Write("Enter second number: ");
        if (!int.TryParse(Console.ReadLine(), out int b)) return;
        Console.Write("Enter third number: ");
        if (!int.TryParse(Console.ReadLine(), out int c)) return;

        int count = 0;
        for (int i = 0; i <= numbers.Length - 3; i++)
        {
            if (numbers[i] == a && numbers[i + 1] == b && numbers[i + 2] == c)
                count++;
        }

        Console.WriteLine("Sequence repetitions: " + count);
    }
}

using System;
using System.Linq;

class Program
{
    static void Main()
    {
        int[] first = { 1, 2, 3, 4, 4, 5 };
        int[] second = { 3, 4, 4, 5, 6, 7 };

        int[] common = first.Intersect(second).ToArray();

        Console.WriteLine("First array: " + string.Join(" ", first));
        Console.WriteLine("Second array: " + string.Join(" ", second));
        Console.WriteLine("Common elements: " + string.Join(" ", common));
    }
}

using System;

class Program
{
    static void Main()
    {
        int[,] numbers =
        {
            { 4, 8, 1 },
            { 9, 2, 7 },
            { 5, 3, 6 }
        };

        int min = numbers[0, 0];
        int max = numbers[0, 0];

        for (int i = 0; i < numbers.GetLength(0); i++)
        {
            for (int j = 0; j < numbers.GetLength(1); j++)
            {
                Console.Write(numbers[i, j] + " ");
                if (numbers[i, j] < min) min = numbers[i, j];
                if (numbers[i, j] > max) max = numbers[i, j];
            }
            Console.WriteLine();
        }

        Console.WriteLine("Minimum: " + min);
        Console.WriteLine("Maximum: " + max);
    }
}

using System;
using System.Linq;

class Program
{
    static void Main()
    {
        Console.Write("Enter sentence: ");
        string sentence = Console.ReadLine() ?? "";
        string[] words = sentence.Split(' ', StringSplitOptions.RemoveEmptyEntries);

        for (int i = 0; i < words.Length; i++)
            words[i] = new string(words[i].Reverse().ToArray());

        Console.WriteLine(string.Join(" ", words));
    }
}

using System;

class Program
{
    static void Main()
    {
        Console.Write("Enter sentence: ");
        string sentence = (Console.ReadLine() ?? "").ToLower();
        string vowels = "aeiouаеєиіїоуюя";
        int count = 0;

        foreach (char symbol in sentence)
        {
            if (vowels.Contains(symbol))
                count++;
        }

        Console.WriteLine("Vowels count: " + count);
    }
}
